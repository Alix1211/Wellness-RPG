def sub(path,pairs):
    s=open(path,encoding='utf8').read()
    for old,new in pairs:
        assert s.count(old)==1,(str(path),old[:90],s.count(old))
        s=s.replace(old,new)
    open(path,'w',encoding='utf8').write(s)
