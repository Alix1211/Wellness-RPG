from pathlib import Path
W=Path(sys.argv[1]);R=Path(sys.argv[2]);D=R/'assets/characters/dialogue-men'
def sub(path,pairs):
    s=open(path,encoding='utf8').read()
    for old,new,*c in pairs:
        n=c[0] if c else 1
        assert s.count(old)==n,(str(path),old[:80],s.count(old))
        s=s.replace(old,new)
    open(path,'w',encoding='utf8').write(s)
